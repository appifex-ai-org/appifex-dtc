    After the existing `signOut()` method, add the complete Apple Sign In implementation (D-02 — hashed-nonce is a security requirement):
    ```swift
    // Phase 4 (FIRE-02 D-02): hashed-nonce flow — security requirement; raw nonce kept, hash sent to Apple
    private var currentNonce: String?

    func signInWithApple() {
        let nonce = randomNonceString()
        currentNonce = nonce
        let request = ASAuthorizationAppleIDProvider().createRequest()
        request.requestedScopes = [.fullName, .email]
        request.nonce = sha256(nonce) // hashed nonce sent to Apple; raw nonce retained in currentNonce
        let controller = ASAuthorizationController(authorizationRequests: [request])
        controller.delegate = self
        controller.performRequests()
    }

    private func randomNonceString(length: Int = 32) -> String {
        var bytes = [UInt8](repeating: 0, count: length)
        _ = SecRandomCopyBytes(kSecRandomDefault, bytes.count, &bytes)
        let charset = Array("0123456789ABCDEFGHIJKLMNOPQRSTUVXYZabcdefghijklmnopqrstuvwxyz-._")
        return String(bytes.map { charset[Int($0) % charset.count] })
    }

    private func sha256(_ input: String) -> String {
        SHA256.hash(data: Data(input.utf8))
            .compactMap { String(format: "%02x", $0) }.joined()
    }
    ```

    if (provider === 'firebase') {
      const plistPath = `${projectDir}/GoogleService-Info.plist`
      let plistContent: string
      try {
        plistContent = await runner.readFile(plistPath)
      } catch {
        // GoogleService-Info.plist not yet present (firebase_provision runs after build wiring)
        // REVERSED_CLIENT_ID injection will be retried at provision time
        return
      }
      const reversedClientIdMatch = plistContent.match(

      const reversedClientId = reversedClientIdMatch?.[1]
      if (!reversedClientId) return

      const projectYmlContent = await runner.readFile(ymlPath)
      const projectYml = yaml.load(projectYmlContent) as Record<string, unknown>

      // Inject URL scheme into the app target's info.plist urlSchemes
      const targets = projectYml['targets'] as Record<string, unknown>
      const appTargetName = Object.keys(targets).find((k) => !k.endsWith('Tests'))
      if (!appTargetName) return

      const appTarget = targets[appTargetName] as Record<string, unknown>
      const info = (appTarget['info'] ?? {}) as Record<string, unknown>
      const existingSchemes = (info['CFBundleURLTypes'] as unknown[]) ?? []
      const schemeAlreadyPresent = JSON.stringify(existingSchemes).includes(reversedClientId)

    }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Template content → generated Swift source | .eta templates produce Swift source files; malicious content in template variables (entity names from BaasSchema) could inject arbitrary Swift code |
| GoogleService-Info.plist content → project.yml | REVERSED_CLIENT_ID extracted from plist is embedded into project.yml; if the plist is tampered, an arbitrary URL scheme could be injected |
| js-yaml parse → project.yml mutation | YAML parse/serialize operates on a file in the output directory; path traversal or YAML bomb inputs are possible |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-02-01 | Tampering | Eta template variable injection (entity.name in Swift) | mitigate | Entity names from BaasSchema are validated by infer-schema.ts at inference time; names are identifiers (letters/numbers only); no shell or Eta escaping needed for Swift identifiers |
| T-04-02-02 | Tampering | REVERSED_CLIENT_ID extraction from plist | mitigate | Use regex with a tight character class for the extracted value (`[^<]+`); the extracted string is used only as a URL scheme value (no shell expansion, no eval path) |
| T-04-02-03 | Denial of Service | js-yaml parse on a malformed/oversized project.yml | accept | project.yml is generated by dtc itself (not user-supplied); size is bounded by the generated app structure; yaml.load is safe on controlled input |
| T-04-02-04 | Information Disclosure | GoogleService-Info.plist in project tree | mitigate | UI-SPEC.md copywriting contract includes "Tip: add GoogleService-Info.plist to your project .gitignore" — executor must emit this tip at provision time (Plan 04) |
</threat_model>
